package com.api.exh.service; import com.api.exh.dto.*; import com.api.exh.integration.jocata.JocataClientHolder; import com.fable.apiclient.Json; import com.fable.apiclient.ApiException; import com.fable.apiclient.jocata.JocataClient; import lombok.extern.slf4j.Slf4j; import org.springframework.http.HttpEntity; import org.springframework.http.HttpHeaders; import org.springframework.http.MediaType; import org.springframework.http.ResponseEntity; import org.springframework.stereotype.Service; import org.springframework.web.client.RestTemplate; import java.util.*; @Service @Slf4j public class JocataService { private final RestTemplate restTemplate; private final String API_URL = "https://api.jocata.com"; public JocataService(RestTemplate restTemplate) { this.restTemplate = restTemplate; } /** * Screen a person/entity name against compliance lists * @param request Name screening request with name and optional details * @return Screening response with match status and details * @throws Exception if API call fails */ public ScreeningResponse screenName( NameScreeningRequest request) throws Exception { try { // Validate input if (request.getName() == null || request.getName().trim().isEmpty()) { throw new IllegalArgumentException("Name is required"); } // Get Jocata client JocataClient jc = JocataClientHolder.get(); // Build request JSON String jsonRequest = buildNameScreeningRequest(request); log.info("Screening name: {}", request.getName()); // Send request (encryption handled internally) String jsonResponse = jc.invoke("/apiscreening/v1", jsonRequest); // Parse response ScreeningResponse response = parseNameScreeningResponse(jsonResponse); log.info("Screening result - Match: {}, Count: {}", response.isMatch(), response.getMatchCount()); return response; } catch (ApiException e) { log.error("Jocata API error [{}]: {}", e.kind(), e.getMessage()); throw new JocataApiException( "Screening failed: " + e.getMessage(), e); } catch (Exception e) { log.error("Unexpected error in name screening", e); throw new JocataApiException( "Unexpected error: " + e.getMessage(), e); } } public ScreeningResponse screenTransaction( TransactionScreeningRequest request) throws Exception { try { // Validate if (request.getTransactionId() == null) { throw new IllegalArgumentException("Transaction ID required"); } // Get client JocataClient jc = JocataClientHolder.get(); // Build request String jsonRequest = buildTransactionScreeningRequest(request); log.info("Screening transaction: {}", request.getTransactionId()); // Send request String jsonResponse = jc.invoke( "/transactionscreening/v1", jsonRequest); // Parse response ScreeningResponse response = parseScreeningResponse(jsonResponse); return response; } catch (Exception e) { log.error("Transaction screening failed", e); throw new JocataApiException( "Transaction screening failed: " + e.getMessage(), e); } } /** * Check if Jocata API is healthy */ public Map<String, Object> healthCheck() throws Exception { try { JocataClient jc = JocataClientHolder.get(); Map<String, Object> health = jc.healthCheck(); log.info("Health check: {}", health.get("status")); return health; } catch (Exception e) { log.error("Health check failed", e); throw new JocataApiException("Health check failed", e); } } // ============================================ // Private Helper Methods // ============================================ private String buildNameScreeningRequest( NameScreeningRequest req) { // Build party object Map<String, Object> partyVO = new LinkedHashMap<>(); partyVO.put("name", req.getName().toUpperCase()); if (req.getDob() != null) { partyVO.put("dob", req.getDob()); } if (req.getNationality() != null) { partyVO.put("nationality", req.getNationality()); } if (req.getCountry() != null) { partyVO.put("country", req.getCountry()); } if (req.getPan() != null) { partyVO.put("pan", req.getPan()); } if (req.getAadhar() != null) { partyVO.put("aadhar", req.getAadhar()); } if (req.getPassport() != null) { partyVO.put("passport", req.getPassport()); } partyVO.put("customerId", req.getCustomerId() != null ? req.getCustomerId() : UUID.randomUUID().toString()); // Build request list VO Map<String, Object> requestListVO = new LinkedHashMap<>(); requestListVO.put("businessUnit", "RAMP"); requestListVO.put("subBusinessUnit", "API"); requestListVO.put("requestType", "API"); requestListVO.put("requestUuId", generateRequestId()); requestListVO.put("requestVOList", Collections.singletonList(partyVO)); // Build root Map<String, Object> rampRequest = new LinkedHashMap<>(); rampRequest.put("listMatchingPayload", Collections.singletonMap("requestListVO", requestListVO)); Map<String, Object> root = new LinkedHashMap<>(); root.put("rampRequest", rampRequest); return Json.write(root); } private String buildTransactionScreeningRequest( TransactionScreeningRequest req) { // Build sender Map<String, Object> senderVO = buildPartyVO( req.getSenderName(), "sender", req.getSenderCustomerType(), req.getSenderDob(), req.getSenderNationality()); // Build receiver Map<String, Object> receiverVO = buildPartyVO( req.getReceiverName(), "receiver", req.getReceiverCustomerType(), req.getReceiverDob(), req.getReceiverNationality()); // Build request list VO Map<String, Object> requestListVO = new LinkedHashMap<>(); requestListVO.put("businessUnit", "RAMP"); requestListVO.put("subBusinessUnit", "API"); requestListVO.put("requestType", "TSG"); requestListVO.put("tranId", req.getTransactionId()); if (req.getAccountNumber() != null) { requestListVO.put("accountNumber", req.getAccountNumber()); } if (req.getRemarks() != null) { requestListVO.put("remarks", req.getRemarks()); } requestListVO.put("requestVOList", Arrays.asList(senderVO, receiverVO)); // Build root Map<String, Object> rampRequest = new LinkedHashMap<>(); rampRequest.put("listMatchingPayload", Collections.singletonMap("requestListVO", requestListVO)); Map<String, Object> root = new LinkedHashMap<>(); root.put("rampRequest", rampRequest); return Json.write(root); vo.put("type", type); vo.put("customerType", customerType); if (dob != null) { vo.put("dob", dob); } if (nationality != null) { vo.put("nationality", nationality); } vo.put("customerId", UUID.randomUUID().toString()); return vo; } private ScreeningResponse parseNameScreeningResponse( String jsonResponse) throws Exception { Map<String, Object> responseMap = Json.parseObject(jsonResponse); Map<String, Object> rampResponse = (Map) responseMap.get("rampResponse"); ScreeningResponse response = new ScreeningResponse(); response.setStatusCode((String) rampResponse.get("statusCode")); response.setStatusMessage((String) rampResponse.get("statusMessage")); if (!"000".equals(response.getStatusCode())) { response.setMatch(false); return response; } Map<String, Object> listMatching = (Map) rampResponse.get("listMatchingResponse"); if (listMatching != null) { List<Map<String, Object>> responseList = (List) listMatching.get("responseListVO"); if (responseList != null && !responseList.isEmpty()) { Map<String, Object> first = responseList.get(0); String matchFlag = (String) first.get("matchFlag"); response.setMatch("Y".equals(matchFlag)); response.setMatchCount((Integer) first.getOrDefault("matchCount", 0)); response.setMatches((List) first.get("matches")); } } return response; } private ScreeningResponse parseScreeningResponse( String jsonResponse) throws Exception { Map<String, Object> responseMap = Json.parseObject(jsonResponse); Map<String, Object> rampResponse = (Map) responseMap.get("rampResponse"); ScreeningResponse response = new ScreeningResponse(); response.setStatusCode((String) rampResponse.get("statusCode")); response.setStatusMessage((String) rampResponse.get("statusMessage")); return response; } private String generateRequestId() { return "IBL-" + System.currentTimeMillis(); } }

package com.api.exh.controller; import com.api.exh.dto.*; import com.api.exh.service.JocataService; import lombok.extern.slf4j.Slf4j; import org.springframework.beans.factory.annotation.Autowired; import org.springframework.http.ResponseEntity; import org.springframework.web.bind.annotation.*; import java.util.Map; @RestController @RequestMapping("/api/jocata") @Slf4j public class JocataController { @Autowired private JocataService jocataService; @PostMapping("/screen-name") public ResponseEntity<?> screenName( @RequestBody NameScreeningRequest request) { try { ScreeningResponse response = jocataService.screenName(request); if (response.isMatch()) { return ResponseEntity .status(403) .body(new ApiResponse( "MATCH_FOUND", "Screening match found", response)); } return ResponseEntity.ok( new ApiResponse("SUCCESS", "No match found", response)); } catch (Exception e) { log.error("Error screening name", e); return ResponseEntity .status(500) .body(new ApiResponse( "ERROR", e.getMessage(), null)); } } @PostMapping("/screen-transaction") public ResponseEntity<?> screenTransaction( @RequestBody TransactionScreeningRequest request) { try { ScreeningResponse response = jocataService.screenTransaction(request); return ResponseEntity.ok(response); } catch (Exception e) { log.error("Error screening transaction", e); return ResponseEntity .status(500) .body(new ApiResponse( "ERROR", e.getMessage(), null)); } } @GetMapping("/health") public ResponseEntity<Map> health() { try { Map<String, Object> health = jocataService.healthCheck(); return ResponseEntity.ok(health); } catch (Exception e) { log.error("Health check failed", e); return ResponseEntity .status(503) .build(); } } }
