Method 2: Using REST Client with Manual Encryption

Advanced Method - Manual control over encryption/decryption

Steps

1 Build JSON Request

Create the request JSON object

2 Encrypt Request

Use JocataClient.encryptEnvelope(jsonRequest)

3 Send Encrypted Request

POST the encrypted envelope to API endpoint

4 Decrypt Response

Decrypt response using JocataClient

Implementation

☕ Manual Encryption Method import org.apache.http.client.HttpClient; import org.apache.http.impl.client.HttpClientBuilder; public class ManualEncryptionExample { public void callJocataAPI() throws Exception { JocataClient jc = JocataClientHolder.get(); // Step 1: Build JSON request String jsonRequest = buildRequestJSON(); // Step 2: Encrypt request String encryptedEnvelope = jc.encryptEnvelope(jsonRequest); // Step 3: Send encrypted request using HTTP client HttpClient httpClient = HttpClientBuilder.create().build(); HttpPost request = new HttpPost("https://api.jocata.com/apiscreening/v1"); request.setHeader("Content-Type", "application/json"); request.setHeader("Authorization", "Bearer YOUR_TOKEN"); request.setEntity(new StringEntity(encryptedEnvelope)); HttpResponse response = httpClient.execute(request); String encryptedResponse = EntityUtils.toString(response.getEntity()); // Step 4: Decrypt response String jsonResponse = jc.decryptResponse(encryptedResponse); System.out.println("Response: " + jsonResponse); } }

Method 3: Using HTTP Client (OkHttp)

Modern HTTP Library - OkHttp with JocataClient encryption

Implementation

☕ OkHttp Implementation import okhttp3.*; public class OkHttpJocataClient { private static final OkHttpClient httpClient = new OkHttpClient.Builder() .connectTimeout(10, TimeUnit.SECONDS) .readTimeout(30, TimeUnit.SECONDS) .build(); public String callNameScreening(String jsonRequest) throws Exception { JocataClient jc = JocataClientHolder.get(); // Encrypt request String encryptedBody = jc.encryptEnvelope(jsonRequest); // Build request Request request = new Request.Builder() .url("https://api.jocata.com/apiscreening/v1") .post(RequestBody.create(encryptedBody, MediaType.parse("application/json"))) .addHeader("Authorization", "Bearer YOUR_TOKEN") .addHeader("Content-Type", "application/json") .build(); // Execute request try (Response response = httpClient.newCall(request).execute()) { if (!response.isSuccessful()) { throw new IOException("Unexpected code " + response); } // Get response body String encryptedResponse = response.body().string(); // Decrypt response String jsonResponse = jc.decryptResponse(encryptedResponse); return jsonResponse; } } }

☕ Spring RestTemplate Service @Service public class JocataApiService { private final RestTemplate restTemplate; private final String API_URL = "https://api.jocata.com"; public JocataApiService(RestTemplate restTemplate) { this.restTemplate = restTemplate; } public String callNameScreening(String jsonRequest) throws Exception { JocataClient jc = JocataClientHolder.get(); // Encrypt request String encryptedBody = jc.encryptEnvelope(jsonRequest); // Set headers HttpHeaders headers = new HttpHeaders(); headers.set("Authorization", "Bearer YOUR_TOKEN"); headers.setContentType(MediaType.APPLICATION_JSON); // Create HTTP entity HttpEntity<String> entity = new HttpEntity<>(encryptedBody, headers); try { // Send POST request ResponseEntity<String> response = restTemplate.postForEntity( API_URL + "/apiscreening/v1", entity, String.class ); // Decrypt response String jsonResponse = jc.decryptResponse(response.getBody()); return jsonResponse; } catch (HttpClientErrorException e) { throw new JocataApiException("API Error: " + e.getMessage(), e); } } }

Complete Name Screening Service

☕ Complete NameScreeningService.java package com.api.exh.service.jocata; import com.fable.apiclient.Json; import com.fable.apiclient.ApiException; import com.fable.apiclient.jocata.JocataClient; import com.api.exh.integration.jocata.JocataClientHolder; import java.util.*; @Service @Slf4j public class NameScreeningService { public ScreeningResponse screenName(ScreeningRequest request) throws Exception { try { // Get JocataClient instance JocataClient jc = JocataClientHolder.get(); // Build request JSON Map<String, Object> jsonPayload = buildNameScreeningRequest(request); String jsonRequest = Json.write(jsonPayload); log.info("Sending screening request: {}", request.getName()); // Send request (encryption handled internally) String jsonResponse = jc.invoke("/apiscreening/v1", jsonRequest); // Parse response ScreeningResponse response = parseNameScreeningResponse(jsonResponse); log.info("Screening completed. Match: {}", response.isMatch()); return response; } catch (ApiException e) { log.error("Jocata API error [{}]: {}", e.kind(), e.getMessage()); throw new ScreeningException("API call failed: " + e.getMessage(), e); } catch (Exception e) { log.error("Unexpected error during screening", e); throw new ScreeningException("Screening failed: " + e.getMessage(), e); } } private Map<String, Object> buildNameScreeningRequest(ScreeningRequest request) { // Build party object Map<String, Object> partyVO = new LinkedHashMap<>(); partyVO.put("name", request.getName().toUpperCase()); if (request.getDob() != null) { partyVO.put("dob", formatDob(request.getDob())); } if (request.getNationality() != null) { partyVO.put("nationality", request.getNationality()); } if (request.getCountry() != null) { partyVO.put("country", request.getCountry()); } if (request.getPan() != null) { partyVO.put("pan", request.getPan()); } if (request.getAadhar() != null) { partyVO.put("aadhar", request.getAadhar()); } partyVO.put("customerId", request.getCustomerId()); // Build request list VO Map<String, Object> requestListVO = new LinkedHashMap<>(); requestListVO.put("businessUnit", "RAMP"); requestListVO.put("subBusinessUnit", "API"); requestListVO.put("requestType", "API"); requestListVO.put("requestUuId", generateRequestId()); requestListVO.put("requestVOList", Collections.singletonList(partyVO)); // Build root request Map<String, Object> rampRequest = new LinkedHashMap<>(); Map<String, Object> listMatching = new LinkedHashMap<>(); listMatching.put("requestListVO", requestListVO); rampRequest.put("listMatchingPayload", listMatching); Map<String, Object> root = new LinkedHashMap<>(); root.put("rampRequest", rampRequest); return root; } private ScreeningResponse parseNameScreeningResponse(String jsonResponse) throws Exception { Map<String, Object> responseMap = Json.parseObject(jsonResponse); Map<String, Object> rampResponse = (Map) responseMap.get("rampResponse"); ScreeningResponse response = new ScreeningResponse(); response.setStatusCode((String) rampResponse.get("statusCode")); response.setStatusMessage((String) rampResponse.get("statusMessage")); // Check if successful if (!"000".equals(response.getStatusCode())) { response.setMatch(false); return response; } // Parse response list Map<String, Object> listMatching = (Map) rampResponse.get("listMatchingResponse"); if (listMatching == null) { response.setMatch(false); return response; } List<Map<String, Object>> responseList = (List) listMatching.get("responseListVO"); if (responseList == null || responseList.isEmpty()) { response.setMatch(false); return response; } Map<String, Object> first = responseList.get(0); String matchFlag = (String) first.get("matchFlag"); response.setMatch("Y".equals(matchFlag)); response.setMatchCount((Integer) first.getOrDefault("matchCount", 0)); response.setMatches((List<Map<String, Object>>) first.get("matches")); return response; } private String formatDob(String dob) { // Convert DD/MM/YYYY to DDMMMYYYY // This is just an example, adjust based on your format return dob.replace("/", ""); } private String generateRequestId() { return "IBL-" + System.currentTimeMillis(); } }

Complete Transaction Screening Service

☕ Complete TransactionScreeningService.java package com.api.exh.service.jocata; import com.fable.apiclient.Json; import com.fable.apiclient.jocata.JocataClient; import com.api.exh.integration.jocata.JocataClientHolder; import java.util.*; @Service @Slf4j public class TransactionScreeningService { public TransactionScreeningResponse screenTransaction( TransactionScreeningRequest request) throws Exception { try { JocataClient jc = JocataClientHolder.get(); // Build request Map<String, Object> jsonPayload = buildTransactionScreeningRequest(request); String jsonRequest = Json.write(jsonPayload); log.info("Screening transaction: {}", request.getTransactionId()); // Send request String jsonResponse = jc.invoke("/transactionscreening/v1", jsonRequest); // Parse response TransactionScreeningResponse response = parseTransactionScreeningResponse(jsonResponse); log.info("Transaction screening completed. Matches: {}", response.getTotalMatches()); return response; } catch (Exception e) { log.error("Transaction screening failed", e); throw new ScreeningException("Transaction screening failed", e); } } private Map<String, Object> buildTransactionScreeningRequest( TransactionScreeningRequest request) { // Build sender Map<String, Object> senderVO = buildPartyVO( request.getSenderName(), "sender", request.getSenderCustomerType(), request.getSenderDob(), request.getSenderNationality() ); // Build receiver Map<String, Object> receiverVO = buildPartyVO( request.getReceiverName(), "receiver", request.getReceiverCustomerType(), request.getReceiverDob(), request.getReceiverNationality() ); // Build request list VO Map<String, Object> requestListVO = new LinkedHashMap<>(); requestListVO.put("businessUnit", "RAMP"); requestListVO.put("subBusinessUnit", "API"); requestListVO.put("requestType", "TSG"); requestListVO.put("tranId", request.getTransactionId()); requestListVO.put("accountNumber", request.getAccountNumber()); requestListVO.put("remarks", request.getRemarks()); requestListVO.put("requestVOList", Arrays.asList(senderVO, receiverVO)); // Build root Map<String, Object> rampRequest = new LinkedHashMap<>(); Map<String, Object> listMatching = new LinkedHashMap<>(); listMatching.put("requestListVO", requestListVO); rampRequest.put("listMatchingPayload", listMatching); Map<String, Object> root = new LinkedHashMap<>(); root.put("rampRequest", rampRequest); return root; } private Map<String, Object> buildPartyVO(String name, String type, String customerType, String dob, String nationality) { Map<String, Object> vo = new LinkedHashMap<>(); vo.put("name", name.toUpperCase()); vo.put("type", type); vo.put("customerType", customerType); if (dob != null) { vo.put("dob", dob); } if (nationality != null) { vo.put("nationality", nationality); } vo.put("customerId", UUID.randomUUID().toString()); return vo; } private TransactionScreeningResponse parseTransactionScreeningResponse( String jsonResponse) throws Exception { Map<String, Object> responseMap = Json.parseObject(jsonResponse); Map<String, Object> rampResponse = (Map) responseMap.get("rampResponse"); TransactionScreeningResponse response = new TransactionScreeningResponse(); response.setStatusCode((String) rampResponse.get("statusCode")); response.setStatusMessage((String) rampResponse.get("statusMessage")); Map<String, Object> listMatching = (Map) rampResponse.get("listMatchingResponse"); if (listMatching != null) { List<Map<String, Object>> responseList = (List) listMatching.get("responseListVO"); if (responseList != null) { response.setTotalMatches(responseList.size()); response.setResponses(responseList); } } return response; } }

⚠️ Error Handling & Response Codes

HTTP Status Codes

Code Meaning Action
200 OK - Request successful Parse response normally
400 Bad Request - Invalid JSON format Check request JSON structure and fields
401 Unauthorized - Invalid credentials Check API token and authentication
403 Forbidden - Access denied Verify API key permissions
404 Not Found - Invalid endpoint Check API endpoint URL
500 Server Error - Gateway error Retry request, contact support if persists
503 Service Unavailable - API down Retry after some time

API Response Status Codes

statusCode statusMessage Meaning
000 Success Request processed successfully, check matchFlag
001 Invalid Request Check JSON structure and required fields
002 Processing Error Screening engine error, retry later
003 Authentication Failed Check API credentials
004 Configuration Error Check api-client.properties configuration

Error Handling Implementation

☕ Comprehensive Error Handling try { String response = jc.invoke("/apiscreening/v1", jsonRequest); ScreeningResponse result = parseResponse(response); // Check API response status if (!"000".equals(result.getStatusCode())) { logger.error("API Error: {} - {}", result.getStatusCode(), result.getStatusMessage()); throw new ScreeningException(result.getStatusMessage()); } // Check match result if (result.isMatch()) { logger.warn("MATCH FOUND: {}", result.getMatches()); // Handle match - escalate, block, etc. } else { logger.info("No match found"); } } catch (ApiException e) { // Handle API exceptions logger.error("API Exception [{}]: {}", e.kind(), e.getMessage()); switch (e.kind()) { case "TimeoutException": logger.error("Request timeout - retrying"); // Implement retry logic break; case "ConnectionException": logger.error("Connection failed - check network/gateway"); break; case "AuthenticationException": logger.error("Authentication failed - check credentials"); break; case "EncryptionException": logger.error("Encryption error - check configuration"); break; default: logger.error("Unknown API error: {}", e.responseBody()); } throw new ScreeningException("Screening failed", e); } catch (Exception e) { logger.error("Unexpected error", e); throw new ScreeningException("Unexpected error: " + e.getMessage(), e); }
